
DATABASE TABLE RELATION NOTES

3. specializations

4. skills

5. certifications

7. career_specialization_weights

- FK: specialization_id -> specializations.specialization_id

8. career_skill_weights
- FK: career_id -> careers.career_id
- FK: skill_id -> skills.skill_id

9. career_certification_weights
- FK: career_id -> careers.career_id
- FK: certification_id -> certifications.certification_id
